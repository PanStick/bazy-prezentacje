for file in /import-data/*.txt; do
    if [ -f "$file" ]; then
        echo "Importuję: $(basename "$file")"
        # TU ZMIEŃ --precision na s, ms lub ns w zależności od testów
        influx write \
            --bucket exercises \
            --org university \
            --token my-super-secret-token-2024 \
            --precision s \
            --file "$file" \
            --skip-verify \
            --http-debug
        if [ $? -eq 0 ]; then
            echo "  -> Sukces: $(basename "$file") zaimportowany."
        else
            echo "  -> Błąd: Nie udało się zaimportować $(basename "$file"). SPRAWDŹ FORMAT DANYCH."
        fi
    fi
done
